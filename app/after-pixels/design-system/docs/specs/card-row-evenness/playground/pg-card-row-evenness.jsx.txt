// Card row evenness rig: a bottom strip with the options, three routes and a live readout. Loads before main.jsx; touches no app module.
(() => {
  const KEY = 'pg_card_row_evenness_v1';
  const OPTS = [
    { n: 0, name: 'One card, no rule', why: 'Both rows use the same card (GsRowCard) and CardRow does nothing more than today: every card stretches to the tallest and the spare height collects at the bottom.', cost: 'Dead space at the bottom of shorter cards, and parts do not line up.' },
    { n: 1, name: 'Shared lines', why: 'CardRow owns four lines (picture, name, blurb, tags) and every card sits on them, so each part starts level across the row.', cost: 'A shorter blurb or set of tags leaves the rest of its line blank. The tallest card in the row sets the lines, even when it is scrolled out of view.' },
    { n: 2, name: 'Fixed shape', why: 'Every card is one shape: name one line, blurb at least two lines, tags on the bottom edge. No card depends on its neighbours.', cost: 'Tags that run to two lines start higher than their neighbours. A one-line blurb leaves one blank line.' },
  ];
  const ROUTES = [
    { n: 1, name: 'Home · Available games', go: () => { window.kitGoState && window.kitGoState('home'); return 'Available games'; } },
    { n: 2, name: 'Daily Word · More games', go: () => { window.kitGoState && window.kitGoState('word-signed-out'); return 'More games'; } },
    { n: 3, name: 'Daily Mystery · More games', note: 'opens Daily Word, then presses the Daily Mystery card', go: () => {
      window.kitGoState && window.kitGoState('word-signed-out');
      setTimeout(() => { const b = [...document.querySelectorAll('.gs-more .gs-titlebtn')].find((x) => x.textContent.trim() === 'Daily Mystery'); if (b) b.click(); }, 700);
      return 'More games';
    } },
  ];
  const load = () => { try { return JSON.parse(localStorage.getItem(KEY)) || {}; } catch (e) { return {}; } };
  const save = (s) => { try { localStorage.setItem(KEY, JSON.stringify(s)); } catch (e) {} };
  const init = load();
  document.documentElement.dataset.pgOpt = String(Math.min(init.opt ?? 1, 2));

  const scroller = (el) => { for (let p = el.parentElement; p; p = p.parentElement) { const s = getComputedStyle(p); if (/(auto|scroll)/.test(s.overflowY) && p.scrollHeight > p.clientHeight) return p; } return document.scrollingElement; };
  const findRow = (title) => [...document.querySelectorAll('.mcp-crow')].find((r) => (r.querySelector('.mcp-crow-title') || {}).textContent === title);
  const ring = (title, tries = 0) => {
    const r = findRow(title);
    if (!r) { if (tries < 30) setTimeout(() => ring(title, tries + 1), 150); return; }
    window.scrollTo(0, 0); const sc = scroller(r); const top = r.getBoundingClientRect().top - (sc === document.scrollingElement ? 0 : sc.getBoundingClientRect().top) + sc.scrollTop - 80;
    sc.scrollTo({ top, behavior: 'auto' });
    r.classList.add('pgc-ring'); setTimeout(() => r.classList.remove('pgc-ring'), 1600);
  };

  // Readout per row on screen: card heights, how far the names start apart, and the most blank space inside one card.
  const parts = (card) => {
    const flat = []; const walk = (el) => { for (const c of el.children) { const d = getComputedStyle(c).display; if (d === 'contents' || c.matches('.gs-feat, .gs-feat-body, .gs-feat-body > .gs-stack-sm')) walk(c); else flat.push(c); } };
    walk(card); return flat;
  };
  const measure = () => [...document.querySelectorAll('.mcp-crow')].map((r) => {
    const cards = [...r.querySelectorAll('.mcp-crow-track > .mcp-card')]; if (!cards.length) return null;
    const hs = cards.map((c) => c.getBoundingClientRect().height);
    const names = cards.map((c) => { const h = c.querySelector('h3'); return h ? h.getBoundingClientRect().top - c.getBoundingClientRect().top : 0; });
    let dead = 0, deadAt = '';
    cards.forEach((c) => {
      const cs = getComputedStyle(c); const inner = c.getBoundingClientRect().height - parseFloat(cs.paddingTop) - parseFloat(cs.paddingBottom) - parseFloat(cs.borderTopWidth) - parseFloat(cs.borderBottomWidth);
      const ps = parts(c); const gap = 12; // the anatomy's one gap; Today's home card uses 16 and 8, so its figure reads a little high
      const used = ps.reduce((a, p) => a + p.getBoundingClientRect().height, 0) + Math.max(0, ps.length - 1) * gap;
      // In options 1 and 2 a part's box can be taller than its text; count that blank too.
      const textBlank = ps.reduce((a, p) => { if (p.tagName !== 'P' && p.tagName !== 'H3') return a; const rg = document.createRange(); rg.selectNodeContents(p); const rs = [...rg.getClientRects()]; if (!rs.length) return a; const tb = Math.max(...rs.map((x) => x.bottom)) - Math.min(...rs.map((x) => x.top)); return a + Math.max(0, p.getBoundingClientRect().height - tb); }, 0);
      const d = Math.max(0, inner - used) + textBlank;
      if (d > dead) { dead = d; const h = c.querySelector('h3'); deadAt = h ? h.textContent.trim() : ''; }
    });
    const spread = (a) => Math.max(...a) - Math.min(...a);
    return { title: (r.querySelector('.mcp-crow-title') || {}).textContent, hs: spread(hs), names: spread(names), dead, deadAt };
  }).filter(Boolean);

  const Strip = () => {
    const [opt, setOpt] = React.useState(Math.min(init.opt ?? 1, 2));
    const [open, setOpen] = React.useState(init.open ?? true);
    const [why, setWhy] = React.useState(false);
    const [read, setRead] = React.useState([]);
    const ref = React.useRef(null);
    React.useEffect(() => { document.documentElement.dataset.pgOpt = String(opt); save({ opt, open }); }, [opt, open]);
    React.useEffect(() => { const t = setInterval(() => setRead(measure()), 600); return () => clearInterval(t); }, []);
    React.useEffect(() => { const el = ref.current; if (!el) return undefined; const ro = new ResizeObserver(() => document.documentElement.style.setProperty('--pgc-h', el.offsetHeight + 'px')); ro.observe(el); return () => ro.disconnect(); }, []);
    const o = OPTS[opt];
    const f = (x) => Math.round(x);
    return (
      <div ref={ref} className="pgc-strip" data-open={String(open)} role="region" aria-label="Playground options">
        <div className="pgc-row">
          <span className="pgc-k">OPTION</span>
          {OPTS.map((x) => <button key={x.n} type="button" className="pgc-b" aria-pressed={opt === x.n} onClick={() => setOpt(x.n)}>{x.n + ' ' + x.name}</button>)}
          <button type="button" className="pgc-b" aria-expanded={why} onClick={() => setWhy(!why)}>{why ? 'Less' : 'Why · cost'}</button>
          <button type="button" className="pgc-b pgc-fold" aria-expanded={open} onClick={() => setOpen(!open)}>{open ? 'Hide' : 'Show'}</button>
        </div>
        {why && <p className="pgc-why"><b>{o.n + ' ' + o.name}.</b> {o.why} <b>Cost:</b> {o.cost}</p>}
        {open && <React.Fragment>
          <div className="pgc-row">
            <span className="pgc-k">GO TO</span>
            {ROUTES.map((r) => <button key={r.n} type="button" className="pgc-b" title={r.note || ''} onClick={() => { const t = r.go(); setTimeout(() => ring(t), r.n === 3 ? 1300 : 500); }}>{r.n + ' ' + r.name}</button>)}
          </div>
          <div className="pgc-read">{read.length ? read.map((r) => (
            <div key={r.title}><b>{r.title}</b>: heights differ by <span className={r.hs > 1 ? 'bad' : ''}>{f(r.hs)}px</span> · names start <span className={r.names > 1 ? 'bad' : ''}>{f(r.names)}px</span> apart · most blank in one card <span className={r.dead > 1 ? 'bad' : ''}>{f(r.dead)}px</span>{r.deadAt ? ' (' + r.deadAt + ')' : ''}</div>
          )) : 'No card row on this page. Use Go to.'}</div>
        </React.Fragment>}
      </div>
    );
  };
  const host = document.createElement('div'); host.id = 'pgc-host';
  const mount = () => { document.body.appendChild(host); ReactDOM.createRoot(host).render(<Strip />); };
  if (document.body) mount(); else document.addEventListener('DOMContentLoaded', mount);
})();
