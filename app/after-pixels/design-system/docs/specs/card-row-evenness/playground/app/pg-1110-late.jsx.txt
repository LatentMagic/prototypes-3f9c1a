// ---- Trial (2026-10-13, not ratified): GsChatScroll, one shared chat picture that scrolls ----
// Opens on its latest turn and scrolls back; the pill says there is more above and takes you there (the home hero's behaviour).
// Default height 290; `fill` stretches it to its box instead. The pill is an arrow only; "Back to the start" is its spoken name (trial).
window.GsChatScroll = ({ turns, label, height = 290, fill = false, more = 'Back to the start' }) => {
  const box = React.useRef(null); const [chat, setChat] = React.useState(null); const [up, setUp] = React.useState(false);
  React.useEffect(() => {
    const b = box.current && box.current.querySelector('.mcp-chat-body'); if (!b) return undefined;
    setChat(b.parentElement);
    b.tabIndex = 0; b.setAttribute('role', 'region'); b.setAttribute('aria-label', label + ', scrolls');
    let moved = false; const mark = () => { const u = b.scrollTop > 4; b.dataset.more = u ? 'up' : ''; setUp(u); };
    const end = () => { if (!moved) b.scrollTop = b.scrollHeight; mark(); };
    const stop = () => { moved = true; };
    b.addEventListener('scroll', mark, { passive: true });
    b.addEventListener('wheel', stop, { passive: true }); b.addEventListener('touchstart', stop, { passive: true }); b.addEventListener('keydown', stop);
    const ro = new ResizeObserver(end); ro.observe(b); [...b.children].forEach((c) => ro.observe(c));
    end(); const t = setTimeout(end, 600);
    return () => { clearTimeout(t); ro.disconnect(); b.removeEventListener('scroll', mark); b.removeEventListener('wheel', stop); b.removeEventListener('touchstart', stop); b.removeEventListener('keydown', stop); };
  }, [turns]);
  const toTop = () => { const b = chat.querySelector('.mcp-chat-body'); b.scrollTo({ top: 0, behavior: window.matchMedia('(prefers-reduced-motion: reduce)').matches ? 'auto' : 'smooth' }); };
  return (
    <div ref={box} className={'gs-chatscroll' + (fill ? ' is-fill' : '')}>
      <GsChatPic turns={turns} width={440} height={height} label={label} />
      {chat && ReactDOM.createPortal(
        <button type="button" className="gs-hero-up" hidden={!up} onClick={toTop} aria-label={more}>
          <DS.Icon name="down" size={16} style={{ transform: 'rotate(180deg)' }} />
        </button>, chat)}
    </div>
  );
};

// The product page's row: three pictures at the shared default height; a longer conversation scrolls instead of growing the row.
window.GsChatRow = ({ id }) => {
  const name = GS_GAMES[id].name;
  return (
    <div className="gs-chatrow" tabIndex={0} role="region" aria-label={'How ' + name + ' looks in your chat'}>
      {(GS_CHATS[id] || []).map((c, i) => (
        <figure key={i} className="gs-chatfig">
          <GsChatScroll turns={c.turns} label={name + ' being played in a chat'} />
          <figcaption className="gs-chatcap">{c.cap}</figcaption>
        </figure>
      ))}
    </div>
  );
};
