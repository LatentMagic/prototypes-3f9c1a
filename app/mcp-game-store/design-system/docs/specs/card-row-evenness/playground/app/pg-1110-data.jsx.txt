// ---- Generated earlier Daily Mystery days, in the real grammar. SAMPLES, not real cases. ----
const PG10_SUS = ['Ivy Marchbank', 'Colonel Rufus Hale', 'Delphine Rooke', 'Arthur Penhallow', 'Mr Graves', 'Sebastian Wyle', 'Dr Helena Strand', 'Nell Okafor'];
const PG10_WPN = ['brass candlestick', 'vial of foxglove', 'antique pistol', 'iron poker', 'curtain cord', 'paper knife', 'marble bust', 'hatpin'];
const PG10_RMS = ['library', 'conservatory', 'butler’s pantry', 'billiard room', 'wine cellar', 'dining room', 'morning room', 'study'];
const pg10Qn = (q) => (q === 0 ? 'No questions' : q === 1 ? '1 question' : q + ' questions');
const pg10Acc = (s, w, r) => s + ', with the ' + w + ', in the ' + r;
PZ.mystery.all.forEach((x) => {
  const i = x.i; if (i === 0 || !x.played || PZ.mystery.keep[i]) return;
  const cs = (i * 3) % 8; const cw = (i * 5 + 1) % 8; const cr = (i * 7 + 2) % 8; const q = i % 3;
  const qs = Array.from({ length: q }, (_, k) => (k === 0
    ? ['Question 1', 'Was ' + PG10_SUS[(i + 1) % 8] + ' in the ' + PG10_RMS[(i + 4) % 8] + '?', i % 2 ? 'Yes' : 'No']
    : ['Question 2', 'Did ' + PG10_SUS[(i + 5) % 8] + ' have the ' + PG10_WPN[(i + 3) % 8] + '?', i % 3 === 1 ? 'Yes' : 'No']));
  // A wrong accusation gets one part wrong: the suspect, the weapon or the room.
  const part = i % 3;
  const acc = x.ok ? pg10Acc(PG10_SUS[cs], PG10_WPN[cw], PG10_RMS[cr])
    : pg10Acc(PG10_SUS[part === 0 ? (cs + 1) % 8 : cs], PG10_WPN[part === 1 ? (cw + 1) % 8 : cw], PG10_RMS[part === 2 ? (cr + 1) % 8 : cr]);
  const sid = 'mystery-e' + i; x.sid = sid;
  GS_SESSIONS[sid] = { kind: 'puzzle', gid: 'mystery', game: 'Daily Mystery', art: 'murder', number: 64 - i, q, hints: 0, date: pzLong(x.d),
    result: x.line, loss: !x.ok, listTitle: 'Every question', share: x.line, figures: [pg10Qn(q), 'No hints'],
    lines: [...qs, ['Accusation', acc, x.ok ? 'Right' : 'Wrong']] };
});

// ---- Share text above the link (brief item 4), line for line. The link line is the prototype's own. ----
const pg10ShareBase0 = window.gsShareBase;
window.gsShareBase = (s) => {
  if (s.share == null) return null;
  const link = 'https://platform.example/' + GS_GAMES[s.gid].route;
  if (s.gid === 'word' && s.number) {
    // A Hint line is not a guess: the tile rows and the count skip it.
    const guesses = s.lines.filter((l) => /^Guess/.test(l[0])).map((l) => l[1]); const answer = s.answer || guesses[guesses.length - 1];
    const h = s.hints || 0;
    const head = 'Daily Word No. ' + s.number + ' · ' + (s.loss ? 'X' : guesses.length) + '/6' + (h ? ' · ' + h + (h === 1 ? ' hint ' : ' hints ') + '💡'.repeat(h) : '');
    return [head, '', ...guesses.map((g) => gsWordRow(g, answer)), link];
  }
  if (s.gid === 'mystery' && s.number) {
    const q = s.q != null ? s.q : s.lines.filter((l) => /^Question/.test(l[0])).length;
    return ['Daily Mystery No. ' + s.number + ' · ' + (s.loss ? 'wrong accusation' : 'solved') + ' · ' + pg10Qn(q).toLowerCase() + (q ? ' ' + '🔎'.repeat(q) : ''), s.loss ? '❌' : '✅', link];
  }
  return pg10ShareBase0(s);
};
// The share card keeps a blank line as a blank line (Daily Word's line after its head).
// It is drawn inside the line above it, so the card's gaps stay one token (4px); Copy still carries the empty line.
window.GsShareText = ({ lines }) => {
  const shown = []; (lines || []).forEach((l) => { if (l === '' && shown.length) shown[shown.length - 1] += '\n'; else shown.push(l); });
  return (
    <DS.Card style={{ gap: 4, justifyItems: 'start' }}>
      {lines ? shown.map((l, i) => <span key={i} className="gs-num-text" style={{ overflowWrap: 'anywhere', whiteSpace: 'pre-line' }}>{/\n$/.test(l) ? l + '\u200b' : l}</span>) : <GsGap />}
    </DS.Card>
  );
};
